# ── Environment ───────────────────────────────────────────────────
module load miniconda3/24.1.2-py310
source activate phylo_tree

# ── Paths ─────────────────────────────────────────────────────────
WORKDIR=/users/PAS2933/diegodilley/PhyloAnalysis
INPUT=$WORKDIR/asvs_for_tree.fasta
ALIGNED=$WORKDIR/asvs_aligned.fasta
OUTDIR=$WORKDIR/Tree_Output
mkdir -p $OUTDIR

# ── Step 1: MUSCLE alignment ──────────────────────────────────────
echo "$(date) — Starting MUSCLE alignment"
muscle -super5 $INPUT -output $ALIGNED -threads 16
echo "$(date) — MUSCLE alignment complete"

# ── Step 2: ModelTest-NG ──────────────────────────────────────────
echo "$(date) — Starting ModelTest-NG"
modeltest-ng \
  --input $ALIGNED \
  --output $OUTDIR/modeltest_results \
  --processes 16
echo "$(date) — ModelTest-NG complete"

# ── Step 3: PhyML ─────────────────────────────────────────────────
echo "$(date) — Starting PhyML"
phyml \
  -i $ALIGNED \
  -m 012345 \
  -f m \
  -v e \
  -a e \
  -c 4 \
  -o tlr \
  -b 100
echo "$(date) — PhyML complete"

echo "All steps finished!"

#===================================================
#Keep an eye on it while its running with code below
#===================================================
squeue -u $USER        # check job status
tail -f phylo_tree_*.log   # watch the log in real time
