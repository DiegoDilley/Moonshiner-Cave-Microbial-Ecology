# ── Environment ───────────────────────────────────────────────────
module load miniconda3/24.1.2-py310
source activate phylo_tree

# ── Paths ─────────────────────────────────────────────────────────
WORKDIR=/users/PAS2933/diegodilley/PhyloAnalysis
INPUT=$WORKDIR/asvs_aligned.fasta
OUTDIR=$WORKDIR/Tree_Output
mkdir -p $OUTDIR

# ── IQ-TREE tree building ─────────────────────────────────────────
echo "$(date) — Starting IQ-TREE"
iqtree \
  -s $INPUT \
  -m GTR+I+G4 \
  -B 1000 \
  -T 16 \
  --prefix $OUTDIR/final_tree
echo "$(date) — IQ-TREE complete"

# ── Save model summary ────────────────────────────────────────────
cat > $OUTDIR/best_model_summary.txt << EOF
================================================
Model Summary
================================================
Date:              $(date)
Input alignment:   $INPUT
Tool:              IQ-TREE 3.1.1
Model:             GTR+I+G4
Bootstraps:        1000 (ultrafast)
Selected by:       ModelTest-NG (BIC & AIC, weight=1.0)
Tree file:         $OUTDIR/final_tree.treefile
================================================
EOF

echo "================================================"
echo "All steps finished!"
echo "Tree file: $OUTDIR/final_tree.treefile"
echo "Model summary: $OUTDIR/best_model_summary.txt"
echo "================================================"

##exit out of nano using keyoboard then
